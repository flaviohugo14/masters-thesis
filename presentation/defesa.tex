\section{Motivação e problema de pesquisa}
%==============================================================================

\begin{frame}{Moeda, território e o duplo movimento}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{quote}
        \footnotesize\itshape
        ``Modelos consagrados, sejam neoclássicos, de causação cumulativa ou de
        insumo-produto, tratam a moeda e os fluxos monetários como exógenos,
        incapazes de explicar as assimetrias regionais.''
      \end{quote}
      \vspace{-0.5em}
      \hfill{\scriptsize\color{cedcinza}CROCCO et al.\ (2006)}

      \vspace{0.6em}
      \small
      \begin{itemize}\setlength{\itemsep}{0.45em}
        \item \az{A promessa.} A digitalização financeira é celebrada como
              mecanismo democratizante do acesso a serviços bancários.
        \item \az{O duplo movimento.} Municípios periféricos enfrentam, ao mesmo
              tempo, retração acelerada da rede física e exclusão digital.
        \item \az{A hipótese de fundo.} A tecnologia não suspende a hierarquia
              financeira herdada --- ela pode reorganizá-la sob nova lógica.
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{alertblock}{O paradoxo a explicar}
        \small
        A maior expansão digital da história do sistema financeiro brasileiro
        conviveu com a persistência --- e, em dimensões-chave, o aprofundamento
        --- da exclusão financeira qualitativa nas periferias.

        \smallskip
        Não se trata de perguntar \emph{se} a digitalização chegou ao território,
        mas o que ela fez com a \emph{natureza} do crédito que ali circula.
      \end{alertblock}

      \begin{block}{O território em 2022}
        \scriptsize
        \begin{tabular}{@{}r@{\ \ }P{0.70\textwidth}@{}}
          \dest{43,8\%} & dos municípios sem agência (2.440 de 5.570)\\[0.2em]
          \dest{25,8\%} & do estoque em 10 cidades, que reúnem 54,5\% do crédito\\[0.2em]
          \dest{10,5\%} & dos municípios com $IRC>1$\\
        \end{tabular}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{-0.4em}
  \fonte{Fonte: ESTBAN/BCB, PIB municipal do IBGE e Pesquisas FEBRABAN (2013--2024).}
\end{frame}

\begin{frame}{Pergunta de pesquisa, objetivo e eixos analíticos}
  \begin{alertblock}{Pergunta de pesquisa}
    Em que medida a digitalização do sistema bancário, ao mudar a geografia da
    prestação de serviços financeiros, \emph{redefine}, \emph{atenua} ou
    \emph{aprofunda} as desigualdades regionais brasileiras?
  \end{alertblock}

  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.325\textwidth}
      \begin{block}{Eixo i --- redes bancárias}
        \small Como a digitalização influencia o fechamento de agências físicas.
        \vspace{0.4em}\\ \az{Desfecho:} $\ln(n_{\text{agências}}+1)$
      \end{block}
    \end{column}
    \begin{column}{0.325\textwidth}
      \begin{block}{Eixo ii --- viés metropolitano}
        \small Concentração de serviços financeiros em polos urbanos já consolidados.
        \vspace{0.4em}\\ \az{Desfechos:} IRC, ICO
      \end{block}
    \end{column}
    \begin{column}{0.325\textwidth}
      \begin{block}{Eixo iii --- qualidade do crédito}
        \small Diferença centro--periferia em disponibilidade, diversidade e
        condições de acesso.
        \vspace{0.4em}\\ \az{Desfecho:} razão IRF/IRE
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.4em}
  \begin{exampleblock}{Objetivo geral}
    \small Investigar como a digitalização do sistema financeiro redefine as
    desigualdades regionais no Brasil, com foco nas transformações qualitativas do
    acesso ao crédito, articulando análise espacial de dados georreferenciados e
    estudos de caso territoriais comparativos.
  \end{exampleblock}
\end{frame}

\begin{frame}{As cinco hipóteses de trabalho}
  \small
  \begin{description}[H5]\setlength{\itemsep}{0.7em}
    \item[\dest{H1}] Existe uma quebra estrutural empiricamente identificável na
      intensidade da digitalização bancária entre 2015 e 2020, com impacto
      detectável sobre indicadores municipais. \hfill{\scriptsize\color{cedcinza}(datação)}
    \item[\dest{H2}] O efeito sobre o fechamento de agências é maior em municípios
      com alta dependência inicial de canais físicos e baixa centralidade
      financeira. \hfill{\scriptsize\color{cedcinza}(eixo i)}
    \item[\dest{H3}] A digitalização aprofunda o viés metropolitano, ampliando a
      divergência dos indicadores de centralidade entre faixas de porte municipal
      após o choque. \hfill{\scriptsize\color{cedcinza}(eixo ii)}
    \item[\dest{H4}] O efeito sobre a qualidade do crédito é assimétrico: serviços
      básicos se difundem, mas os financiamentos de longo prazo se concentram nos
      centros. \hfill{\scriptsize\color{cedcinza}(eixo iii)}
    \item[\dest{H5}] Parte substancial do efeito sobre o crédito periférico opera
      \emph{através} do fechamento de agências --- canal de mediação entre
      tecnologia e território. \hfill{\scriptsize\color{cedcinza}(mecanismo)}
  \end{description}
\end{frame}

\section{Referencial teórico}
%==============================================================================

\begin{frame}{Moeda, hierarquia financeira e desenvolvimento regional}
  \begin{block}{A cadeia teórica}
    \scriptsize
    \begin{tabular}{@{}P{0.20\textwidth}P{0.24\textwidth}P{0.46\textwidth}@{}}
      \toprule
      \textbf{Elo} & \textbf{Referência} & \textbf{Proposição} \\
      \midrule
      Economia monetária de produção &
        \small Keynes (1973); Carvalho (1992) &
        A moeda não é neutra: sob incerteza, o agente pode reter riqueza na forma
        líquida em vez de produzir.\\[0.35em]
      Preferência pela liquidez regional &
        \small Dow (1982); Chick e Dow (1988) &
        Na periferia, a incerteza informacional é maior; bancos retêm ativos
        líquidos e contraem o crédito longo.\\[0.35em]
      Lugar central nas finanças &
        \small Christaller (1966); Parr e Budd (2000) &
        Serviços complexos se concentram hierarquicamente; serviços básicos se
        dispersam.\\[0.35em]
      Drenagem de liquidez e distância funcional &
        \small Amado (2006); Crocco (2010, 2012); Alessandrini et al.\ (2009) &
        A poupança periférica migra para os centros e retorna como crédito em
        piores condições.\\
      \bottomrule
    \end{tabular}
  \end{block}

  \vspace{-0.2em}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{exampleblock}{Reestruturação bancária: a evidência}
        \scriptsize
        \textbf{Leyshon et al.\ (2008):} fechamento de $-25\%$ em áreas pobres
        contra $-16{,}9\%$ em áreas ricas no Reino Unido.\\
        \textbf{Nguyen (2019):} cada fechamento reduz o crédito a pequenas
        empresas em US\$ 453 mil/ano, com efeito local persistente por até 6 anos.
      \end{exampleblock}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{exampleblock}{No Brasil}
        \scriptsize
        \textbf{Fonseca e Matray (2024):} a entrada de agências eleva depósitos,
        crédito, emprego e empreendedorismo, com mais força em desertos bancários.\\
        \textbf{Schuster (2023):} onde a agência rural fecha, a alternativa
        raramente é digital.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{A dialética da digitalização e a lacuna a preencher}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{A promessa}
        \small
        Plataformas digitais ampliam o alcance de serviços financeiros básicos
        (pagamentos, saques, transferências) e reduzem custos de operação.

        \smallskip
        No Brasil, Pix, \emph{Mobile Banking} e correspondentes ampliam a
        bancarização elementar mesmo onde não há agência.
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{A contrapartida}
        \small
        Operações de maior valor agregado seguem dependendo de provedores nos
        grandes centros (Hashimoto et al., 2022): a lógica do lugar central
        persiste em chave digital.

        \smallskip
        Forma-se uma ``periferia digital'' (Guerra-Leal et al., 2021;
        Malaquias, 2020), excluída até da bancarização elementar.
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.3em}
  \begin{exampleblock}{A lacuna identificada}
    \footnotesize
    \begin{itemize}\setlength{\itemsep}{0.25em}
      \item A tradição pós-keynesiana brasileira documentou o peso da concentração
            espacial sobre a preferência pela liquidez regional, mas \alert{não
            incorporou a digitalização} como variável capaz de mudar a natureza da
            intermediação.
      \item A evidência internacional sobre a assimetria territorial da substituição
            de canais \alert{nunca havia sido testada em escala municipal no Brasil},
            com indicadores que separem \emph{volume} de \emph{qualidade} do crédito.
      \item Limitações metodológicas na aplicação de econometria espacial e
            subexploração dos efeitos diferenciados por tipo de serviço financeiro.
    \end{itemize}
  \end{exampleblock}
\end{frame}

\begin{frame}{Os indicadores construídos: separando volume de qualidade}
  \footnotesize\setlength{\tabcolsep}{4pt}
  \begin{tabular}{@{}P{0.15\textwidth}P{0.44\textwidth}P{0.35\textwidth}@{}}
    \toprule
    \textbf{Sigla} & \textbf{Definição} & \textbf{Leitura} \\

    \addlinespace
    \multicolumn{3}{@{}l}{\az{Centralidade e concentração}}\\
    IRC & Quociente locacional do crédito sobre o PIB & Centralidade creditícia\\
    ICO & HHI das modalidades de crédito & Diversidade do portfólio local\\

    \addlinespace
    \multicolumn{3}{@{}l}{\az{Qualidade e condições do crédito}}\\

              curto prazo (conta 161) & \dest{Proxy central de qualidade}\\
    IRD, IRR, IRL & Quocientes locacionais de depósitos, risco e lucro bancário &
                    Distribuição espacial de depósitos, risco e rentabilidade\\
    \bottomrule
  \end{tabular}

  \vspace{0.4em}
  \begin{alertblock}{Por que a razão IRF/IRE é o indicador-chave}
    \footnotesize Valores altos indicam um portfólio voltado a operações estruturadas e
    relacionais; valores baixos, um crédito predominantemente curto e padronizado.
    É o termômetro que permite testar se a digitalização mexeu no \emph{volume} ou
    na \emph{natureza} do crédito --- a pergunta que a literatura nacional ainda
    não havia respondido.
  \end{alertblock}
\end{frame}

\begin{frame}{Fato estilizado 1 --- o salto digital é recente e abrupto}
  \begin{columns}[c]
    \begin{column}{0.60\textwidth}
      \centering
      \includegraphics[width=\linewidth]{fig-transacoes-1.pdf}
    \end{column}
    \begin{column}{0.38\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{0.45em}
        \item O \emph{Mobile Banking} ultrapassa o \emph{Internet Banking} como
              canal predominante em 2015--2016.
        \item O volume de transações móveis \emph{per capita} dispara a partir de
              2020, com a pandemia e o Pix.
        \item A série é \alert{nacional}: há variação temporal, mas nenhuma
              variação espacial no tratamento.
      \end{itemize}
      \vspace{0.4em}
      \begin{alertblock}{A restrição que define a estratégia}
        \scriptsize
        As Pesquisas FEBRABAN são a única fonte sistemática sobre intensidade de
        digitalização bancária no país e não têm desagregação subnacional. Sem
        variação espacial, estratégias clássicas de painel com tratamento
        heterogêneo ficam descartadas.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Fato estilizado 2 --- a geografia bancária é desigual}
  \begin{columns}[c]
    \begin{column}{0.54\textwidth}
      \centering
      \includegraphics[height=0.74\textheight]{fig-mapa-brasil-1.pdf}
    \end{column}
    \begin{column}{0.44\textwidth}
      \small
      \begin{block}{Municípios sem agência (2022)}
        \centering
        \begin{tabular}{@{}lr@{}}
          \toprule
          \textbf{Região} & \textbf{\% sem agência}\\
          \midrule
          Nordeste & \dest{59,8\%}\\
          Norte    & 40,7\%\\
          Sul      & 36,0\%\\
          Sudeste  & 33,6\%\\
          \midrule
          \textbf{Brasil} & \textbf{43,8\%}\\
          \bottomrule
        \end{tabular}
      \end{block}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{0.25em}
        \item Apenas 14 municípios reuniam 100 ou mais agências; 1.660 eram
              atendidos por uma ou duas.
        \item Os 10 maiores concentravam 25,8\% do estoque e 54,5\% do saldo de
              crédito.

\begin{frame}{O desafio de identificação e a solução \emph{shift-share}}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{O problema}
        \small O tratamento (intensidade da digitalização) varia no tempo, mas é
        idêntico para todos os municípios. Sem variação \emph{cross-section}, não
        há grupo de controle.
      \end{alertblock}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{A solução (Borusyak et al., 2024)}
        \small Desenho \emph{shift-share} à la Bartik: multiplica-se o
        \emph{shifter} nacional por uma medida \emph{cross-section} de exposição
        \emph{ex-ante}. A uniformidade espacial vira vantagem analítica.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.2em}
  \begin{exampleblock}{Especificação geral}
    \vspace{-0.9em}
    \[
      Y_{it} \;=\; \alpha_i \;+\; \lambda_t \;+\;
      \beta\,\bigl(\text{Post}_t \times \ex\bigr) \;+\;
      \gamma X_{it} \;+\; \varepsilon_{it}
    \]
    \vspace{-1.1em}
    \begin{center}
      {\scriptsize\color{cedcinza}
       efeitos fixos de município ($\alpha_i$) e de ano ($\lambda_t$)
       \quad$\cdot$\quad $\text{Post}_t = \mathbb{1}\{t \geq 2020\}$
       \quad$\cdot$\quad controle $\ln(\text{PIB real})$
       \quad$\cdot$\quad erros-padrão clusterizados por município}
    \end{center}
  \end{exampleblock}

  \vspace{-0.2em}
  \begin{block}{As três camadas do Capítulo 3}
    \scriptsize
    \begin{tabular}{@{}P{0.16\textwidth}P{0.79\textwidth}@{}}
      \az{Quantitativa} & Datação do choque (Bai-Perron, Quandt-Andrews, Chow)
        $\cdot$ exposição \emph{ex-ante} por ACP $\cdot$ \emph{event study} e DiD
        contínuo $\cdot$ SLX-DiD $\cdot$ mediação causal $\cdot$ quantílica espacial\\
      \az{Qualitativa} & Quatro estudos de caso territoriais em Minas Gerais,
        selecionados a partir dos resíduos e dos efeitos heterogêneos\\
      \az{Normativa} & Tradução dos coeficientes em simulações contrafactuais e
        recomendações regulatórias\\
    \end{tabular}
  \end{block}
\end{frame}

\begin{frame}{\emph{Spillovers} espaciais (SLX-DiD): vasos comunicantes}
  \vspace{-0.4em}
  \begin{center}
    $Y_{it} = \alpha_i + \lambda_t + \beta\,(\text{Post}_t \times \ex)
      + \textcolor{cedlaranja}{\boldsymbol{\theta\,(\text{Post}_t \times W\ex)}}
      + \gamma X_{it} + \varepsilon_{it}$
  \end{center}
  \vspace{-0.4em}
  {\centering\scriptsize\color{cedcinza}
   $W$ = contiguidade rainha, \emph{row}-normalizada (malha IBGE 2020 via
   \texttt{geobr}; 5.568 municípios em subgrafo único)
   \quad$\cdot$\quad $\beta$ = efeito direto, $\theta$ = \emph{spillover}\par}

  \vspace{0.5em}
  \centering
  \small
  \begin{tabular}{@{}lcc P{0.40\textwidth}@{}}
    \toprule
    \textbf{Desfecho} & \textbf{Direto} ($\beta$) & \textbf{\emph{Spillover}} ($\theta$)
      & \textbf{Interpretação}\\
    \midrule
    $\ln(n_{\text{agências}})$ & \dest{$-2{,}2\%$} & \pos{$+2{,}7\%$} &
      o município perde a agência; a vizinhança a ganha\\
    IRC --- centralidade & \dest{negativo} & \pos{positivo} &
      a centralidade migra para os polos adjacentes\\
    ICO --- concentração & \dest{positivo} & \pos{negativo} &
      o portfólio local se estreita; o dos vizinhos se diversifica\\
    IRF/IRE --- qualidade & \dest{fortemente negativo} & \pos{positivo} &
      o crédito longo sai da periferia e reaparece no entorno\\
    \bottomrule
  \end{tabular}

  \vspace{0.6em}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{Sinais opostos nos quatro desfechos}
        \scriptsize A digitalização não eliminou agências, centralidade,
        diversidade ou crédito longo do território. Ela os \emph{reorganizou no
        espaço}, transferindo-os dos municípios mais vulneráveis para suas
        vizinhanças mais centrais.
      \end{alertblock}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Por que isso explica o ``zero'' do IRC}
        \scriptsize $|\theta|$ é da mesma ordem de $|\beta|$, às vezes maior. O
        coeficiente médio do DiD não espacial é uma mistura de efeitos diretos e
        indiretos de sinais opostos: parte do efeito direto negativo era cancelada
        por um \emph{spillover} positivo.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Implicações e conclusões}

\begin{frame}{Três simulações contrafactuais}
  \small
  \begin{block}{Cenário 1 --- redução de 50\% no fechamento de agências}
    \footnotesize
    Um município com $\ex = +1$ perderia $0{,}8\%$ em vez de $1{,}6\%$ do log do
    estoque. É alívio, não reversão. E o SLX acrescenta: o fechamento não é só perda
    líquida, é \emph{realocação} --- a agência fecha em um lugar e reabre alguns
    quilômetros adiante. \hfill\az{$\Rightarrow$ frear a concentração espacial na origem.}
  \end{block}
  \begin{block}{Cenário 2 --- homogeneização da exposição digital}
    \footnotesize
    Reduziria a vulnerabilidade, mas não eliminaria o problema. O que determina a
    resiliência da cobertura não é só conectividade: é a densidade do mercado
    bancário local. Betim tinha acesso digital pleno e perdeu 36\% das agências;
    Juiz de Fora, com metade do PIB, perdeu 21\%.
    \hfill\az{$\Rightarrow$ inclusão digital sozinha não resolve exclusão financeira.}
  \end{block}
  \begin{block}{Cenário 3 --- piso de uma agência por município}
    \footnotesize
    A mediação responde pela negativa: o \emph{suppression effect} mostra que a
    digitalização reduz o IRF/IRE por um canal que não passa pela quantidade de
    agências. Almenara é o caso-limite --- 5 agências mantidas e queda de 43\% na
    razão. \hfill\az{$\Rightarrow$ o que muda não é a presença da agência, é o que a agência faz.}
  \end{block}
\end{frame}

\begin{frame}{Implicações regulatórias}
  \small
  \begin{enumerate}\setlength{\itemsep}{0.9em}
    \item \az{Migrar a regulação de quantidade de agências para funções financeiras
      mantidas no território.}\\
      \footnotesize Manter a agência não garante manter o crédito de longo prazo. O
      fechamento pode ser compensado por correspondentes e canais digitais
      assistidos, desde que tenham escopo funcional equivalente. Os modelos de
      \emph{service obligation} dos setores postal e de telecomunicações são
      referência adaptável --- calibrada por tempo de deslocamento e vulnerabilidade
      ($\ex$), não por quilômetros em linha reta.
    \item \az{Calibrar os instrumentos de crédito conforme a heterogeneidade
      espacial.}\\
      \footnotesize O SLX mostra que o crédito se desloca no espaço; a quantílica,
      que o efeito depende do ponto de partida de cada município. Uma política
      uniforme produz resultados distintos em Almenara (bancos públicos sustentam a
      função) e em Betim (crédito privado curto, função financeira rasa).
      Instrumentos possíveis: percentual mínimo de carteira em modalidades longas
      para municípios abaixo de certo porte e redesconto condicionado.
    \item \az{Coordenar regulação bancária, infraestrutura digital e desenvolvimento
      regional.}\\
      \footnotesize Conectividade digital não substitui presença institucional
      (Betim) e presença institucional não impede a deterioração do crédito
      (Almenara). O problema é estrutural: exige que Banco Central, BNDES e bancos
      estaduais, ANATEL e o Ministério da Integração atuem de forma articulada.
  \end{enumerate}
\end{frame}

\begin{frame}{Balanço das cinco hipóteses}
  \centering
  \scriptsize
  \begin{tabular}{@{}cP{0.30\textwidth}P{0.44\textwidth}c@{}}
    \toprule
    & \textbf{O que previa} & \textbf{O que a evidência mostrou} & \textbf{Veredito}\\
    \midrule
    \dest{H1} & Quebra estrutural identificável entre 2015 e 2020 &
      2020 confirmado como inflexão dominante. O choque não é um ponto único: é uma
      sequência de rupturas (2014--16, 2018--19, 2020--21) & \pos{Sustentada}\\
    \addlinespace
    \dest{H2} & Efeito maior onde havia alta dependência de canais físicos &
      Direção confirmada, forma desmentida: a contração é não monotônica,
      concentrada nas duas pontas da distribuição de porte & \dest{Reformulada}\\
    \addlinespace
    \dest{H3} & A digitalização aprofunda o viés metropolitano &
      A mais fortalecida. Não é só declínio diferenciado: é transferência
      territorial identificável (direto e \emph{spillover} com sinais opostos) &
      \pos{Sustentada\,$+$}\\
    \addlinespace
    \dest{H4} & Efeito assimétrico sobre a qualidade do crédito &
      Confirmada. Efeito médio nulo sobre centralidade, mas o crédito longo se
      concentra nos centros e recua nas periferias & \pos{Sustentada}\\
    \addlinespace
    \dest{H5} & O canal de transmissão é o fechamento de agências &
      \emph{Suppression effect}. O elo agência $\rightarrow$ crédito longo se
      confirma, mas o canal dominante é o tecnológico & \dest{Reformulada}\\
    \bottomrule
  \end{tabular}

  \vspace{0.5em}
  \begin{alertblock}{A tese que as cinco convergem para sustentar}
    \footnotesize A digitalização financeira \alert{não é territorialmente neutra}.
    O aprendizado, porém, está menos no fato de quatro hipóteses terem se sustentado
    e mais nos pontos em que a evidência obrigou a abandonar a linearidade: o
    fechamento não segue a dependência física ponto a ponto, a deterioração do
    crédito não vem de um canal só, e o empobrecimento das periferias não é
    subtração pura --- o que sai delas reaparece nos centros.
  \end{alertblock}
\end{frame}

\begin{frame}{Contribuições e limitações}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{Quatro contribuições}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{0.35em}
          \item \az{Base de dados.} Painel de indicadores financeiros municipais
            2009--2023, integrando fontes antes usadas separadamente.
          \item \az{Analítica.} Decomposição do efeito em canal direto, canal
            mediado e \emph{spillovers} espaciais, combinada à decomposição
            quantílica.
          \item \az{Empírica.} Os efeitos variam com porte, posição na hierarquia
            urbana e base institucional de cada lugar.
          \item \az{Aplicada.} Mapeamento entre coeficientes estimados e
            instrumentos de política.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{Quatro limitações assumidas}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{0.35em}
          \item \az{Dados.} A série FEBRABAN é nacional. Não há dados municipais de
            juros, prazos ou inadimplência. A ESTBAN registra saldos, não operações
            novas: a direção da causalidade na margem permanece ambígua.
          \item \az{Identificação temporal.} Quebras em 2014--16 e 2018--19
            coincidem com rupturas institucionais que uma data única não separa.
          \item \az{Unidade espacial.} A função financeira transborda o limite
            municipal.
          \item \az{Interpretação qualitativa.} O contexto institucional dos casos
            se apoia em fontes secundárias.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.1em}
  \begin{exampleblock}{Agenda de pesquisa}
    \scriptsize Incorporar dados desagregados do SCR/BCB $\cdot$ mudar a escala
    para microrregiões ou regiões funcionais $\cdot$ investigar os mecanismos
    institucionais com métodos qualitativos $\cdot$ separar o choque digital das
    rupturas institucionais contemporâneas.
  \end{exampleblock}
\end{frame}

\begin{frame}{Considerações finais}
  \begin{block}{}
    \centering
    {\large\az{Moeda e crédito não são variáveis exógenas ao desenvolvimento
     regional. São forças que estruturam a geografia das oportunidades.}}\\[0.4em]
    \small A digitalização financeira não suspendeu essa dinâmica:
    \emph{reorganizou-a}.
  \end{block}

  \vspace{0.2em}
  \footnotesize
  \begin{itemize}\setlength{\itemsep}{0.35em}
    \item \az{A rede física recuou seletivamente} e a convergência intermunicipal em
      cobertura, que vinha ocorrendo até 2019, congelou a partir de 2020.
    \item \az{A qualidade do crédito se descolou do volume} --- a razão IRF/IRE caiu
      em quase toda a distribuição, com intensidade máxima onde o crédito longo já
      era escasso.
    \item \az{O efeito é um deslocamento, não um desaparecimento} --- efeito direto e
      \emph{spillover} têm sinais opostos: o que sai da periferia reaparece nos
      polos adjacentes.
    \item \az{O canal dominante é tecnológico, não mecânico} --- a transformação
      qualitativa da carteira não depende de a agência fechar; ela ocorre também com
      a agência aberta.
  \end{itemize}

  \vspace{0.4em}
  \begin{alertblock}{}
    \centering\small Cabe à pesquisa e à política pública entender como --- para que
    a promessa de inclusão não vire, na prática, mais um vetor de desigualdade.
  \end{alertblock}
\end{frame}
